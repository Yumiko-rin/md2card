def fibonacci(n: int) -> list[int]:
    """生成斐波那契数列前 n 项"""
    seq = [0, 1]
    while len(seq) < n:
        seq.append(seq[-1] + seq[-2])
    return seq[:n]

print("Hello, md2card!")
